Q. 다음 C언어로 구현된 프로그램을 분석하여 그 실행 결과를 쓰시오.

```c
#include <stdio.h> 

struct Node { 
    struct Node* next;
    unsigned int x; 
};

int main() {
    struct Node t1 = { 0, 5u };
    struct Node t2 = { 0, 7u };
    struct Node t3 = { 0, 11u };

    t3.next = &t2; 
    t2.next = &t1; 

    struct Node* curr = &t3; 
    unsigned int sum = 0;

    while (curr) { 
        sum = sum * 3 + curr->x; 
        curr = curr->next; 
    }

    sum = (sum ^ 42u ) + 100u; 
    printf("%u\n", sum); 
}
```

> **정답: 187**
